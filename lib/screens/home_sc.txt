// ─── All: Classic + Customized seamlessly in one scroll ───────────────────────

class _AllCombinedGrid extends StatelessWidget {
  final List<Map<String, dynamic>> regularItems;
  final List<Map<String, dynamic>> customizedItems;
  final ScrollController scrollController;
  final ValueChanged<Map<String, dynamic>> onRegularTap;

  const _AllCombinedGrid({
    Key? key,
    required this.regularItems,
    required this.customizedItems,
    required this.scrollController,
    required this.onRegularTap,
  }) : super(key: key);

  @override
  Widget build(BuildContext context) {
    final List<Map<String, dynamic>> allItems = [
      ...regularItems.map((e) => {...e, '_type': 'regular'}),
      ...customizedItems.map((e) => {...e, '_type': 'customized'}),
    ];

    final List<Widget> rows = [];

    for (int i = 0; i < allItems.length; i += 2) {
      final leftItem = allItems[i];
      final rightItem = (i + 1 < allItems.length) ? allItems[i + 1] : null;

      Widget buildCell(Map<String, dynamic> food) {
        if (food['_type'] == 'regular') {
          return Expanded(
            child: FoodCardWidget(
              imagePath: food['image'],
              foodName: food['name'],
              subtitle: food['subtitle'],
              rating: food['rating'],
              foodData: food,
              onTap: () => onRegularTap(food),
            ),
          );
        } else {
          return Expanded(
            child: CustomizedFoodCardWidget(
              imagePath: food['image'],
              foodName: food['name'],
              subtitle: food['subtitle'],
              basePrice: food['basePrice'],
              toppings: food['toppings'] as List<ToppingItem>?,
              sideOptions: food['sideOptions'] as List<SideItem>?,
              foodData: food,
            ),
          );
        }
      }

      if (rows.isNotEmpty) {
        rows.add(const SizedBox(height: 14)); // vertical gap
      }

      rows.add(
        SizedBox(
          height: 253,
          child: Row(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              buildCell(leftItem),
              rightItem != null
                  ? buildCell(rightItem)
                  : const Expanded(child: SizedBox.shrink()),
            ],
          ),
        ),
      );
    }

    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 5),
      child: ListView(
        controller: scrollController,
        padding: const EdgeInsets.only(bottom: 200),
        children: rows,
      ),
    );
  }
}

// ─── Classic food grid ────────────────────────────────────────────────────────

class _RegularGrid extends StatelessWidget {
  final List<Map<String, dynamic>> items;
  final ScrollController scrollController;
  final ValueChanged<Map<String, dynamic>> onTap;

  const _RegularGrid({
    Key? key,
    required this.items,
    required this.scrollController,
    required this.onTap,
  }) : super(key: key);

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 5),
      child: GridView.builder(
        controller: scrollController,
        padding: const EdgeInsets.only(bottom: 200),
        gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
          crossAxisCount: 2,
          mainAxisSpacing: 14,
          crossAxisSpacing: 1,
          childAspectRatio: 0.71,
        ),
        itemCount: items.length,
        itemBuilder: (context, index) {
          final food = items[index];
          return FoodCardWidget(
            imagePath: food['image'],
            foodName: food['name'],
            subtitle: food['subtitle'],
            rating: food['rating'],
            foodData: food,
            onTap: () => onTap(food),
          );
        },
      ),
    );
  }
}
